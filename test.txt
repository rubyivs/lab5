curl -s -D - -o /dev/null http://127.0.0.1:3000/418

//GET: неправильний URL
curl -i http://127.0.0.1:3000/abc

// POST: перевірка недозволеного методу
curl -i -X POST http://127.0.0.1:3000/418

// PUT: збереження картинки
curl -i -X PUT --data-binary "@cache/418.jpg" -H "Content-Type: image/jpeg" http://127.0.0.1:3000/200

// GET: отримання картинки після PUT
curl -s -D - -o 200.jpg http://127.0.0.1:3000/200

// PUT: перевірка порожнього тіла запиту
curl -i -X PUT http://127.0.0.1:3000/202

// DELETE: видалення картинки
curl -i -X DELETE http://127.0.0.1:3000/200

curl -i -X DELETE http://127.0.0.1:3000/200
